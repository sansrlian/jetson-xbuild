cmake_minimum_required(VERSION 3.25)
project(jxb_isaac_test LANGUAGES CXX CUDA)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_buffer REQUIRED)
find_package(cuda_buffer REQUIRED)
find_package(isaac_ros_tensor_rt REQUIRED)
find_package(CUDAToolkit REQUIRED)

add_executable(cuda_buffer_check src/cuda_buffer_check.cpp src/fill.cu)
target_compile_features(cuda_buffer_check PRIVATE cxx_std_17)
# Exported targets where the packages provide them, classic variables otherwise.
foreach(dep rclcpp rosidl_buffer cuda_buffer isaac_ros_tensor_rt)
  if(${dep}_TARGETS)
    target_link_libraries(cuda_buffer_check ${${dep}_TARGETS})
  else()
    target_include_directories(cuda_buffer_check PRIVATE ${${dep}_INCLUDE_DIRS})
    target_link_libraries(cuda_buffer_check ${${dep}_LIBRARIES})
  endif()
endforeach()
target_link_libraries(cuda_buffer_check CUDA::cudart)
install(TARGETS cuda_buffer_check DESTINATION lib/${PROJECT_NAME})
ament_package()
